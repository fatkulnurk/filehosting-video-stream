```php
<?php
namespace App\Services;

use App\Models\Directory;
use Illuminate\Support\Str;

class DirectoryService
{
    public function create(string $name, ?int $parentId, int $userId): Directory
    {
        $parent = $parentId
            ? Directory::where('user_id', $userId)->findOrFail($parentId)
            : null;

        $path = $parent ? "{$parent->path}/{$name}" : "/{$name}";

        return Directory::create([
            'user_id' => $userId,
            'parent_id' => $parentId,
            'name' => $name,
            'slug' => Str::slug($name),
            'path' => $path,
        ]);
    }

    public function delete(Directory $directory): void
    {
        foreach ($directory->files as $file) {
            \Storage::disk($file->disk)->delete($file->path);
            $file->delete();
        }
        foreach ($directory->children as $child) {
            $this->delete($child);
        }
        $directory->delete();
    }

    public function getTree(int $userId): \Illuminate\Support\Collection
    {
        return Directory::where('user_id', $userId)
            ->whereNull('parent_id')
            ->with('childrenRecursive')
            ->get();
    }
}
```

```php
<?php
namespace App\Services;

use App\Models\File;
use Illuminate\Support\Str;
use Illuminate\Http\UploadedFile;

class FileService
{
    public function store(UploadedFile $file, ?int $directoryId, int $userId, ?string $diskName = null): File
    {
        $fileModel = File::create([
            'user_id' => $userId,
            'directory_id' => $directoryId,
            'name' => pathinfo($file->getClientOriginalName(), PATHINFO_FILENAME),
            'slug' => Str::slug(pathinfo($file->getClientOriginalName(), PATHINFO_FILENAME)),
        ]);

        // Attach file to Spatie Media Library
        $fileModel->addMedia($file)
            ->usingName($file->getClientOriginalName())
            ->toMediaCollection('default');

        return $fileModel;
    }

    public function delete(File $file): void
    {
        $file->delete(); // Cascades to media via Spatie
    }
}
```

```php
<?php
namespace App\Services;

use App\Enums\StorageDriver;
use App\Models\StorageServer;
use Illuminate\Support\Facades\Config;

class StorageService
{
    /**
     * Register a storage disk from StorageServer model
     * Supports: local, s3, dropbox, gcs, azure, ftp, sftp, rackspace, wasabi, b2
     */
    public function registerDisk(StorageServer $server): void
    {
        $driver = StorageDriver::from($server->driver);

        $config = match ($driver) {
            StorageDriver::Local => [
                'driver' => 'local',
                'root' => $server->config['root'] ?? storage_path("app/{$server->name}"),
            ],
            's3' => [
                'driver' => 's3',
                'key' => $server->config['key'],
                'secret' => $server->config['secret'],
                'region' => $server->config['region'],
                'bucket' => $server->config['bucket'],
                'endpoint' => $server->config['endpoint'] ?? null,
                'use_path_style_endpoint' => $server->config['use_path_style'] ?? false,
            ],
            'dropbox' => [
                'driver' => 'dropbox',
                'token' => $server->config['token'],
            ],
            'gcs' => [
                'driver' => 'gcs',
                'keyFile' => $server->config['key_file'],
                'bucket' => $server->config['bucket'],
            ],
            'azure' => [
                'driver' => 'azure',
                'account-name' => $server->config['account'],
                'account-key' => $server->config['key'],
                'container' => $server->config['container'],
            ],
            'ftp' => [
                'driver' => 'ftp',
                'host' => $server->config['host'],
                'port' => $server->config['port'] ?? 21,
                'username' => $server->config['username'],
                'password' => $server->config['password'],
                'root' => $server->config['root'] ?? '/',
                'passive' => $server->config['passive'] ?? true,
                'ssl' => $server->config['ssl'] ?? false,
            ],
            'sftp' => [
                'driver' => 'sftp',
                'host' => $server->config['host'],
                'port' => $server->config['port'] ?? 22,
                'username' => $server->config['username'],
                'password' => $server->config['password'] ?? null,
                'privateKey' => $server->config['private_key'] ?? null,
                'root' => $server->config['root'] ?? '/',
                'passphrase' => $server->config['passphrase'] ?? null,
            ],
            'rackspace' => [
                'driver' => 'rackspace',
                'username' => $server->config['username'],
                'key' => $server->config['api_key'],
                'container' => $server->config['container'],
                'region' => $server->config['region'],
            ],
            'wasabi' => [
                'driver' => 's3',
                'key' => $server->config['key'],
                'secret' => $server->config['secret'],
                'region' => $server->config['region'],
                'bucket' => $server->config['bucket'],
                'endpoint' => 'https://s3.' . $server->config['region'] . '.wasabisys.com',
                'use_path_style_endpoint' => true,
            ],
            'b2' => [
                'driver' => 'b2',
                'key_id' => $server->config['key_id'],
                'application_key' => $server->config['application_key'],
                'bucket' => $server->config['bucket'],
            ],
            default => throw new \InvalidArgumentException(
                "Unsupported driver: {$server->driver}"
            ),
        };

        Config::set("filesystems.disks.{$server->name}", $config);
    }

    /**
     * Register all active storage disks
     */
    public function registerAllDisks(): void
    {
        StorageServer::where('is_active', true)
            ->each(fn ($server) => $this->registerDisk($server));
    }

    /**
     * Get storage statistics for user
     */
    public function getStats(int $userId): array
    {
        return StorageServer::where('user_id', $userId)
            ->get()
            ->map(fn ($server) => [
                'id' => $server->id,
                'name' => $server->name,
                'driver' => $server->driver,
                'used' => $server->storage_used,
                'limit' => $server->storage_limit,
                'files' => $server->files()->count(),
            ])
            ->toArray();
    }

    /**
     * Update storage used size
     */
    public function syncUsedSize(StorageServer $server): void
    {
        $totalSize = $server->files()->sum('size');
        $server->update(['storage_used' => $totalSize]);
    }
}
```
